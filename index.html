9. Palindrome Using Function 
pal<-function(n){
 rev=0; temp=n
 while(temp>0){
  r=temp%%10
  rev=rev*10+r
  temp=temp%/%10
 }
 return(rev==n)
}

num<-as.integer(readline())
print(pal(num))
