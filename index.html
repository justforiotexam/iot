3. Armstrong Number
num <- as.integer(readline(prompt="Enter number: ")) digits <- as.numeric(strsplit(as.character(num), "")[[1]]) power <- length(digits) sum_val <- sum(digits^power) if (sum_val == num) { print("Armstrong Number") } else { print("Not Armstrong Number") }

9. Palindrome Using Function
palindrome <- function(n) { rev <- 0 temp <- n while (temp > 0) { r <- temp %% 10 rev <- rev * 10 + r temp <- temp %/% 10 } return(rev == n) } num <- as.integer(readline(prompt="Enter number: ")) if (palindrome(num)) { print("Palindrome Number") } else { print("Not a Palindrome Number") }
